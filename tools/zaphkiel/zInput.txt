     2. Run this list, then open zaphkiel_out.txt plus signatures.txt.
     3. tools/zaphkiel/apply_output.py turns those into an updated
        signatures.json for this mod.
   ------------------------------------------------------------------------ */

/* --- 3. Struct offsets -------------------------------------------------- */
/* Every "offsets" key in signatures.json starts UNPROVEN: the value is a     */
/* candidate, and Resolver::offset() will not hand it out until a live probe  */
/* confirms it. So a struct dump here is not a formality -- it is the only    */
/* way to know whether the candidate was ever right.                          */
/*
   Already self-proving, no Zaphkiel needed (geometric validation works):
     struct<Actor><0x228>    -> actor_state_vector
     struct<Actor><0x230>    -> actor_aabb_component
   These three are NOT self-proving and are the current blocker:
     struct<Actor><0x600>    -> actor_level, actor_dimension
     struct<Dimension><0x200>-> dimension_block_source
   Dump the whole object rather than one offset: a struct<> call shows what is
   actually at 0x1F0, whereas asking for 0x1F0 shows only that something is
   there. See docs/ZAPHKIEL-REQUEST-actor-layout.md for the full request.
*/
struct<Actor><0x600>
struct<Dimension><0x200>

/* --- 4. Confirming the entity pipeline --------------------------------- */
/* The actor list is the backbone of entity highlighting.                 */
struct<ActorManager><0x8>

/* --- 5. Dimension -> block source -------------------------------------- */
/* Covered by the struct<Dimension><0x200> above. Kept as a note: this offset's
   failure is silent in one direction (a wrong block source just returns no
   blocks) and a SIGSEGV in the other, which is what the 2026-11-13 tombstone
   showed. Never dereference it on an unproven value.                      */

/* --- 6. Sanity check that the class names used by the UI still exist ---- */
findclass<Zombie>
findclass<Player>
findclass<ItemActor>
